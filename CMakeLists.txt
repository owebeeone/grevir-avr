cmake_minimum_required(VERSION 3.20)
project(grevir_avr VERSION 0.1.0 LANGUAGES CXX)
foreach(dependency IN ITEMS base registers core)
  if(NOT TARGET grevir::${dependency})
    find_package(grevir-${dependency} CONFIG REQUIRED)
  endif()
endforeach()
add_library(grevir_avr INTERFACE)
add_library(grevir::avr ALIAS grevir_avr)
set_target_properties(grevir_avr PROPERTIES EXPORT_NAME avr)
target_compile_features(grevir_avr INTERFACE cxx_std_23)
target_link_libraries(grevir_avr INTERFACE grevir::base grevir::registers grevir::core)
target_include_directories(grevir_avr INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
  $<INSTALL_INTERFACE:include>)
include(GNUInstallDirs)
install(TARGETS grevir_avr EXPORT GrevirAVRTargets)
install(DIRECTORY src/ DESTINATION include FILES_MATCHING PATTERN "*.hpp" PATTERN "*.h")
install(EXPORT GrevirAVRTargets NAMESPACE grevir::
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/grevir-avr)
install(FILES cmake/grevir-avr-config.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/grevir-avr)
install(FILES LICENSE.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/grevir-avr)
option(GREVIR_BUILD_COMPILE_CHECKS "Compile native headers and template users" OFF)
option(GREVIR_BUILD_HOST_TESTS "Run deterministic native behavior tests" OFF)
if(GREVIR_BUILD_HOST_TESTS)
  enable_testing()
endif()
if(GREVIR_BUILD_COMPILE_CHECKS OR GREVIR_BUILD_HOST_TESTS)
  add_subdirectory(tests)
endif()
